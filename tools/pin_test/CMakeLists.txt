cmake_minimum_required(VERSION 3.13)
if(NOT DEFINED ENV{PICO_SDK_PATH})
  set(ENV{PICO_SDK_PATH} /home/bit/pico/pico-sdk)
endif()
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(pin_test C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(pin_test main.c)
pico_enable_stdio_usb(pin_test 1)
pico_enable_stdio_uart(pin_test 0)
target_compile_definitions(pin_test PRIVATE PICO_STDIO_USB_CONNECT_WITHOUT_DTR=1)
target_link_libraries(pin_test pico_stdlib hardware_gpio)
# 不调用 pico_add_extra_outputs：它会尝试联网拉 picotool；UF2 用现成的 picotool 转
