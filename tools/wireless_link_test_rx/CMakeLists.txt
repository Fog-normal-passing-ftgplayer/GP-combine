cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(wireless_link_test_rx C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
pico_sdk_init()

add_executable(wireless_link_test_rx main.c)
pico_generate_pio_header(wireless_link_test_rx ${CMAKE_CURRENT_LIST_DIR}/ws2812.pio)
pico_enable_stdio_usb(wireless_link_test_rx 1)
pico_enable_stdio_uart(wireless_link_test_rx 0)
target_compile_definitions(wireless_link_test_rx PRIVATE PICO_STDIO_USB_CONNECT_WITHOUT_DTR=1)
target_link_libraries(wireless_link_test_rx pico_stdlib hardware_spi hardware_gpio hardware_pio)
pico_add_extra_outputs(wireless_link_test_rx)
