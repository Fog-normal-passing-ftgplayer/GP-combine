cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(receiver_diag C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
pico_sdk_init()

add_executable(receiver_diag main.c)
pico_generate_pio_header(receiver_diag ${CMAKE_CURRENT_LIST_DIR}/ws2812.pio)
pico_enable_stdio_usb(receiver_diag 1)
pico_enable_stdio_uart(receiver_diag 0)
target_compile_definitions(receiver_diag PRIVATE PICO_STDIO_USB_CONNECT_WITHOUT_DTR=1)
target_link_libraries(receiver_diag pico_stdlib pico_multicore hardware_spi hardware_gpio hardware_pio hardware_flash)
pico_add_extra_outputs(receiver_diag)
